Load every page of users

The API returns 20 users per page. A generator fetches page after page until a page comes back empty.

fromAsync collects them all into one list.

async function* getUsers() {
  let page = 1;
  while (true) {
    const res = await fetch(`/api/users?page=${page}`);
    const users = await res.json();
    if (users.length === 0) return;
    yield* users;
    page++;
  }
}

const allUsers = await Array.fromAsync(getUsers());
// [{ id: 1, … }, { id: 2, … }, …]
